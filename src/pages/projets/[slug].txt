    params: { slug: projet.slug },
    props: {
      projet,
      prevProject: projetsAvecImages[index - 1] || null,
      nextProject: projetsAvecImages[index + 1] || null,
    },
  }));
}

function varyColor(hexColor: string, index: number): string {
  // Convertir hex en HSL
  const hex = hexColor.replace("#", "");
  const r = parseInt(hex.substring(0, 2), 16) / 255;
  const g = parseInt(hex.substring(2, 4), 16) / 255;
  const b = parseInt(hex.substring(4, 6), 16) / 255;

  const max = Math.max(r, g, b);
  const min = Math.min(r, g, b);
  let h = 0,
    s = 0,
    l = (max + min) / 2;

  if (max !== min) {
    const d = max - min;
    s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
    switch (max) {
      case r:
        h = ((g - b) / d + (g < b ? 6 : 0)) / 6;
        break;
      case g:
        h = ((b - r) / d + 2) / 6;
        break;
      case b:
        h = ((r - g) / d + 4) / 6;
        break;
    }
  }

  // Variations basées sur l'index
  const variations = [
    { s: -0.05, l: 0.03 },
    { s: 0.05, l: -0.02 },
    { s: -0.03, l: -0.04 },
    { s: 0.08, l: 0.05 },
  ];
  const v = variations[index % variations.length];

  const newS = Math.max(0, Math.min(1, s + v.s));
  const newL = Math.max(0, Math.min(1, l + v.l));

  return `hsl(${Math.round(h * 360)}, ${Math.round(newS * 100)}%, ${Math.round(newL * 100)}%)`;
}

<script>
  // Parallaxe images
  const items = document.querySelectorAll<HTMLElement>(".projet-item");
  const hero = document.querySelector<HTMLElement>(".projet-hero");
  const heroImg = hero?.querySelector("img");
  const heroAmplitude = parseFloat(hero?.dataset.parallaxHero || "-100");

  const currentTransforms = new Map<HTMLElement, number>();
  items.forEach((item) => currentTransforms.set(item, 0));
  let heroCurrentY = 0;

  function lerp(start: number, end: number, factor: number): number {
    return start + (end - start) * factor;
  }

  function updateParallax() {
    const viewportHeight = window.innerHeight;
    // const scrollY = window.scrollY;

    // Parallaxe hero
    if (heroImg && hero) {
      const heroRect = hero.getBoundingClientRect();
      const heroProgress = -heroRect.top / viewportHeight;
      const heroTargetY = heroProgress * heroAmplitude;
      heroCurrentY = lerp(heroCurrentY, heroTargetY, 0.1);
      heroImg.style.transform = `translateY(${heroCurrentY}px)`;
    }

    items.forEach((item) => {
      const rect = item.getBoundingClientRect();
      const amplitude = parseFloat(item.dataset.parallax || "0");

      // Calcul du progress : 0 quand l'élément entre en bas, 1 quand il sort en haut
      const elementCenter = rect.top + rect.height / 2;
      const progress = 1 - elementCenter / viewportHeight;

      // Cible de translation
      const targetY = progress * amplitude;

      // Interpolation pour un mouvement fluide
      const currentY = currentTransforms.get(item) || 0;
      const newY = lerp(currentY, targetY, 0.1);
      currentTransforms.set(item, newY);

      item.style.transform = `translateY(${newY}px)`;
    });

    requestAnimationFrame(updateParallax);
  }

  // Démarre l'animation
  requestAnimationFrame(updateParallax);
</script>
